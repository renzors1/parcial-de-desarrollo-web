using BlazorApp1.Components;
using BlazorApp1.Components.Services;

var builder = WebApplication.CreateBuilder(args);

// Servicios de componentes y render interactivo
builder.Services.AddRazorComponents()
    .AddInteractiveServerComponents();

// Cliente HTTP configurado para la API de Spring Boot
builder.Services.AddHttpClient("SpringBoot", client =>
{
    client.BaseAddress = new Uri("http://localhost:9999/");
    client.Timeout = TimeSpan.FromSeconds(3);
});

builder.Services.AddScoped(sp => sp.GetRequiredService<IHttpClientFactory>().CreateClient("SpringBoot"));

// Inyección de dependencias de negocio y estado
builder.Services.AddScoped<BackendStatusService>();
builder.Services.AddScoped<TokenService>();
builder.Services.AddScoped<AuthService>();
builder.Services.AddScoped<LibroService>();
builder.Services.AddScoped<VentaService>();
builder.Services.AddScoped<CarritoService>();
builder.Services.AddScoped<LicenciaDigitalService>();
builder.Services.AddScoped<PromocionService>();
builder.Services.AddScoped<EditorialService>();
builder.Services.AddScoped<CalificacionService>();

var app = builder.Build();

if (!app.Environment.IsDevelopment())
{
    app.UseExceptionHandler("/Error", createScopeForErrors: true);
    app.UseHsts();
}
app.UseStatusCodePagesWithReExecute("/not-found", createScopeForStatusCodePages: true);
app.UseHttpsRedirection();

app.UseAntiforgery();

app.MapStaticAssets();
app.MapRazorComponents<App>()
    .AddInteractiveServerRenderMode();

app.Run();

@code {
    private string? mensajeConexion = null;

    protected override void OnInitialized()
    {
        TokenService.OnChange += StateHasChanged;
        BackendStatus.OnChange += StateHasChanged;
        CarritoService.OnChange += StateHasChanged;
    }

    public void Dispose()
    {
        TokenService.OnChange -= StateHasChanged;
        BackendStatus.OnChange -= StateHasChanged;
        CarritoService.OnChange -= StateHasChanged;
    }

    private async Task ProbarConexion()
    {
        mensajeConexion = "Verificando...";
        StateHasChanged();
        var ok = await BackendStatus.ProbarConexionBackendAsync();
        mensajeConexion = ok ? "Conectado a API" : "Servidor no detectado";
        _ = Task.Delay(3000).ContinueWith(_ =>
        {
            mensajeConexion = null;
            InvokeAsync(StateHasChanged);
        });
    }
}

    private List<LibroDto> todosLosLibros = new();
    private List<EditorialDto> editorialesDisponibles = new();
    private List<CalificacionUsuarioDto> listaCalificaciones = new();
    private List<string> categoriasDisponibles = new();
    private PromocionDto? promocionGeneral = null;

    private bool cargando = true;
    private string? errorCarga = null;
    private string terminoBusqueda = "";
    private string categoriaSeleccionada = "";
    private string editorialSeleccionada = "";
    private string? mensajeCompra = null;
    private HashSet<long> librosAgregadosFeedback = new();

    private List<LibroDto> librosFiltrados => todosLosLibros
        .Where(l => string.IsNullOrWhiteSpace(terminoBusqueda) ||
                    l.Titulo.Contains(terminoBusqueda, StringComparison.OrdinalIgnoreCase) ||
                    l.Isbn.Contains(terminoBusqueda, StringComparison.OrdinalIgnoreCase) ||
                    l.AutorPrincipal.Contains(terminoBusqueda, StringComparison.OrdinalIgnoreCase))
        .Where(l => string.IsNullOrWhiteSpace(categoriaSeleccionada) ||
                    l.Categoriases.Any(c => c.NombreCategoria.Equals(categoriaSeleccionada, StringComparison.OrdinalIgnoreCase)) ||
                    l.CategoriaPrincipal.Equals(categoriaSeleccionada, StringComparison.OrdinalIgnoreCase))
        .Where(l => string.IsNullOrWhiteSpace(editorialSeleccionada) ||
                    l.NombreEditorial.Equals(editorialSeleccionada, StringComparison.OrdinalIgnoreCase))
        .ToList();

    protected override async Task OnInitializedAsync()
    {
        CarritoService.OnChange += StateHasChanged;
        await CargarDatos();
    }

    public void Dispose()
    {
        CarritoService.OnChange -= StateHasChanged;
    }

    private async Task CargarDatos()
    {
        cargando = true;
        errorCarga = null;
        try
        {
            todosLosLibros = await LibroService.GetAllAsync();
            editorialesDisponibles = await EditorialService.GetAll();
            try { listaCalificaciones = await CalificacionService.GetAll(); } catch { }

            var cats = todosLosLibros
                .SelectMany(l => l.Categoriases.Select(c => c.NombreCategoria))
                .Where(c => !string.IsNullOrWhiteSpace(c))
                .Distinct()
                .ToList();

            if (cats.Count == 0)
            {
                cats = new List<string> { "Programación", "Arquitectura", "Cloud Computing", "Desarrollo Web", "Base de Datos" };
            }
            categoriasDisponibles = cats;

            var activas = await PromocionService.GetActivas();
            promocionGeneral = activas.FirstOrDefault();
        }
        catch (Exception ex)
        {
            errorCarga = $"No se pudo sincronizar el catálogo: {ex.Message}";
        }
        finally
        {
            cargando = false;
        }
    }

    private (decimal Rating, int Resenas) ObtenerMetricas(long? libroId)
    {
        var id = libroId ?? 1;
        var reviews = listaCalificaciones.Where(c => c.LibroId == id).ToList();
        if (reviews.Count > 0)
        {
            return (Math.Round((decimal)reviews.Average(c => c.Puntuacion), 1), reviews.Count);
        }
        var determinista = Math.Round(4.4m + ((id % 5) * 0.1m), 1);
        var count = 18 + (int)((id * 7) % 45);
        return (determinista, count);
    }

    private void LimpiarFiltros()
    {
        terminoBusqueda = "";
        categoriaSeleccionada = "";
        editorialSeleccionada = "";
    }

    private void ComprarLibro(LibroDto libro)
    {
        CarritoService.AgregarLibro(libro);
        mensajeCompra = $"'{libro.Titulo}' se agregó a tu bolsa de compras.";
        if (libro.Id.HasValue)
        {
            var id = libro.Id.Value;
            librosAgregadosFeedback.Add(id);
            _ = Task.Run(async () =>
            {
                await Task.Delay(1500);
                librosAgregadosFeedback.Remove(id);
                await InvokeAsync(StateHasChanged);
            });
        }
    }

@code {
    private string nombreTitular = "";
    private string correoCliente = "";
    private string ultimoComprobante = "";
    private List<LicenciaDigitalDto> licenciasGeneradas = new();
    private string? mensajeCopiado = null;
    private LicenciaDigitalDto? licenciaLectura = null;
    private bool procesando = false;
    private bool compraExitosa = false;
    private string? errorMensaje = null;
    private List<PromocionDto> promocionesActivas = new();

    private decimal BaseImponible => Math.Round(CarritoService.TotalPagar / 1.18m, 2);
    private decimal IgvCalculado => CarritoService.TotalPagar - BaseImponible;

    protected override async Task OnInitializedAsync()
    {
        CarritoService.OnChange += StateHasChanged;
        
        if (TokenService.IsAuthenticated && TokenService.CurrentUser != null)
        {
            var user = TokenService.CurrentUser;
            nombreTitular = $"{user.FirstName} {user.LastName}".Trim();
            if (string.IsNullOrWhiteSpace(nombreTitular))
            {
                nombreTitular = user.Login ?? "Cliente";
            }
            correoCliente = user.Email ?? "";
        }

        try
        {
            promocionesActivas = await PromocionService.GetActivas();
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error al cargar promociones: {ex.Message}");
        }
    }

    public void Dispose()
    {
        CarritoService.OnChange -= StateHasChanged;
    }

    private void AplicarPromocion(PromocionDto promo)
    {
        CarritoService.AplicarPromocion(promo);
    }

    private void RemoverPromocion()
    {
        CarritoService.RemoverPromocion();
    }

    private async Task CopiarClave(string codigo)
    {
        try
        {
            await JS.InvokeVoidAsync("navigator.clipboard.writeText", codigo);
            mensajeCopiado = $"Clave DRM '{codigo}' copiada al portapapeles con éxito.";
        }
        catch (Exception)
        {
            mensajeCopiado = $"Clave DRM: {codigo}";
        }
    }

    private void AbrirLector(LicenciaDigitalDto lic)
    {
        licenciaLectura = lic;
    }

    private async Task ProcesarCheckout()
    {
        if (string.IsNullOrWhiteSpace(nombreTitular))
        {
            errorMensaje = "Por favor ingresa tu Nombre y Apellidos.";
            return;
        }

        if (string.IsNullOrWhiteSpace(correoCliente) || !correoCliente.Contains("@"))
        {
            errorMensaje = "Por favor ingresa un Correo Electrónico válido para emitir tu comprobante y licencias digitales.";
            return;
        }

        procesando = true;
        errorMensaje = null;

        try
        {
            var totalPagar = CarritoService.TotalPagar;
            var subtotalBase = Math.Round(totalPagar / 1.18m, 2);
            var igv = Math.Round(totalPagar - subtotalBase, 2);
            var descuentoTotal = Math.Round(CarritoService.MontoDescuento, 2);
            var correlativo = DateTime.UtcNow.ToString("yyMMddHHmmss");
            var comprobante = $"B001-{correlativo}";
            var promoId = CarritoService.PromocionAplicada?.Id;
            var itemsCopia = CarritoService.Items.ToList();

            var venta = new VentaDto
            {
                NumeroComprobante = comprobante,
                Cliente = $"{nombreTitular} ({correoCliente})",
                FechaVenta = DateTime.UtcNow,
                SubtotalVenta = subtotalBase,
                TotalDescuento = descuentoTotal,
                Igv = igv,
                TotalVenta = totalPagar,
                UsuarioCreacion = TokenService.CurrentUser?.Login ?? correoCliente,
                FechaCreacion = DateTime.UtcNow,
                Detalles = itemsCopia.Select(i =>
                {
                    var descItem = CarritoService.PromocionAplicada != null
                        ? Math.Round(i.Subtotal * (decimal)(CarritoService.PromocionAplicada.PorcentajeDescuento / 100.0), 2)
                        : 0m;
                    return new DetalleVentaDto
                    {
                        LibroId = i.Libro.Id!.Value,
                        LibroTitulo = i.Libro.Titulo,
                        CantidadVenta = i.Cantidad,
                        PrecioUnitario = i.Libro.Precio,
                        Descuento = descItem,
                        Subtotal = i.Subtotal - descItem,
                        PromocionAplicadaId = promoId
                    };
                }).ToList()
            };

            // 1. Descontar cupo si aplica
            if (promoId.HasValue)
            {
                await PromocionService.DescontarCupo(promoId.Value, 1);
            }

            // 2. Registrar venta
            await VentaService.RegistrarVentaAsync(venta);

            // 3. Generar licencias digitales (tantas como la cantidad comprada de cada libro)
            ultimoComprobante = comprobante;
            licenciasGeneradas = await LicenciaService.GenerarLicenciasParaVenta(itemsCopia, correoCliente);

            // 4. Limpiar carrito y marcar éxito
            CarritoService.Limpiar();
            compraExitosa = true;
        }
        catch (Exception ex)
        {
            errorMensaje = $"Ocurrió un error al procesar tu compra: {ex.Message}";
        }
        finally
        {
            procesando = false;
        }
    }
}

        ? misLicencias
        : misLicencias.Where(l =>
            (l.Libro?.Titulo?.Contains(filtroTexto, StringComparison.OrdinalIgnoreCase) ?? false) ||
            (l.Libro?.AutorPrincipal?.Contains(filtroTexto, StringComparison.OrdinalIgnoreCase) ?? false) ||

            listaCalificaciones = await CalificacionService.GetAll();

            // fallback
        }
    }

    private async Task ConsultarLicenciasInvitado()
    {
        if (string.IsNullOrWhiteSpace(correoOConsulta))
        {
            errorCarga = "Por favor ingresa tu correo o nombre para consultar tus licencias.";
            return;
        }

        consultando = true;
        errorCarga = null;
        try
        {
            misLicencias = await LicenciaService.GetMisLicencias(correoOConsulta.Trim());
            haConsultado = true;
        }
        catch (Exception ex)
        {
            errorCarga = $"Error al consultar licencias: {ex.Message}";
        }
        finally
        {
            consultando = false;
        }
    }

    private async Task ConsultarRapido(string demoUser)
    {
        correoOConsulta = demoUser;
        await ConsultarLicenciasInvitado();
    }

    private async Task HandleKeyUpConsulta(KeyboardEventArgs e)
    {
        if (e.Key == "Enter")
        {
            await ConsultarLicenciasInvitado();
        }
    }

    private void LimpiarConsulta()
    {
        correoOConsulta = "";
        haConsultado = false;
    }

    private async Task CargarLicencias()
    {
        cargando = true;
        errorCarga = null;
        try
        {
            misLicencias = await LicenciaService.GetMisLicencias();
        }
        catch (Exception ex)
        {
            errorCarga = $"No se pudieron obtener las licencias: {ex.Message}";
        }
        finally
        {
            cargando = false;
        }
    }

    private decimal ObtenerPuntuacion(long? libroId)
    {
        var id = libroId ?? 1;
        var list = listaCalificaciones.Where(c => c.LibroId == id).ToList();
        if (list.Count > 0)
        {
            return Math.Round((decimal)list.Average(c => c.Puntuacion), 1);
        }
        // Valoración determinista en catálogo
        return Math.Round(4.4m + ((id % 5) * 0.1m), 1);
    }

    private async Task CopiarCodigo(string codigo)
    {
        try
        {
            await JS.InvokeVoidAsync("navigator.clipboard.writeText", codigo);
            mensajeCopiado = $"Clave DRM copiada al portapapeles: {codigo}";
        }
        catch (Exception)
        {
            mensajeCopiado = $"Clave DRM: {codigo}";
        }
    }

    private void AbrirLector(LicenciaDigitalDto lic)

        comentarioCalificacion = "";
        calificacionEnviada = false;
    }

    private async Task EnviarCalificacion()

                : $"[{licenciaSeleccionada.Libro?.Titulo}] {comentarioCalificacion}";

            var ok = await CalificacionService.RegistrarOpinion(calificacion, comentario, cliente, licenciaSeleccionada.Libro?.Id);

                mensajeCopiado = "¡Tu reseña y puntuación han sido enviadas y guardadas exitosamente!";
                listaCalificaciones = await CalificacionService.GetAll();
            }
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error al enviar calificación: {ex.Message}");
        }
        finally
        {
            enviandoCalificacion = false;
        }
    }
}

@code {
    private List<LibroDto> libros = new();
    private List<EditorialDto> editoriales = new();
    private LibroDto libroActual = new();
    private LibroDto? libroAEliminar = null;
    private string campoAutorTexto = "";
    private long editorialSeleccionadaId = 0;
    private string categoriaSeleccionada = "Programación";
    private bool cargando = true;
    private bool guardando = false;
    private bool mostrarModal = false;
    private bool mostrarModalEliminar = false;
    private string? mensajeAlerta;

    protected override async Task OnInitializedAsync()
    {
        await CargarLibros();
        try
        {
            editoriales = await EditorialService.GetAll();
        }
        catch
        {
            // fallback
        }
    }

    private async Task CargarLibros()
    {
        cargando = true;
        libros = await LibroService.GetAll();
        cargando = false;
    }

    private void AbrirModalNuevo()
    {
        libroActual = new LibroDto
        {
            Precio = 50.00m,
            Idioma = "Español",
            Activo = true,
            PortadaUrl = "/covers/cover-clean-arch.svg"
        };
        campoAutorTexto = "";
        editorialSeleccionadaId = editoriales.FirstOrDefault()?.Id ?? 0;
        categoriaSeleccionada = "Programación";
        mostrarModal = true;
    }

    private void AbrirModalEditar(LibroDto libro)
    {
        libroActual = new LibroDto
        {
            Id = libro.Id,
            Isbn = libro.Isbn,
            Titulo = libro.Titulo,
            Sinopsis = libro.Sinopsis,
            Precio = libro.Precio,
            Stock = libro.Stock,
            Idioma = libro.Idioma,
            Activo = libro.Activo,
            PortadaUrl = libro.PortadaUrl,
            Editorial = libro.Editorial,
            Categoriases = new List<CategoriaDto>(libro.Categoriases),
            Autoreses = new List<AutorDto>(libro.Autoreses)
        };

        campoAutorTexto = libro.AutorPrincipal;
        editorialSeleccionadaId = libro.Editorial?.Id ?? 0;
        categoriaSeleccionada = libro.CategoriaPrincipal ?? "Programación";
        mostrarModal = true;
    }

    private void CerrarModal()
    {
        mostrarModal = false;
    }

    private async Task GuardarLibro()
    {
        guardando = true;

        if (string.IsNullOrWhiteSpace(libroActual.Isbn))
        {
            libroActual.Isbn = $"978-{DateTime.UtcNow.Ticks.ToString()[^10..]}";
        }

        // Separación rigurosa de Nombres y Apellidos para evitar concatenaciones duplicadas
        var autorLimpio = (campoAutorTexto ?? "").Trim();
        if (!string.IsNullOrWhiteSpace(autorLimpio))
        {
            var parts = autorLimpio.Split(' ', 2, StringSplitOptions.RemoveEmptyEntries);
            var nombres = parts.Length > 0 ? parts[0] : "";
            var apellidos = parts.Length > 1 ? parts[1] : "";

            if (libroActual.Autoreses.Count > 0)
            {
                libroActual.Autoreses[0].Nombres = nombres;
                libroActual.Autoreses[0].Apellidos = apellidos;
            }
            else
            {
                libroActual.Autoreses.Add(new AutorDto { Nombres = nombres, Apellidos = apellidos });
            }
        }

        // Asignar editorial seleccionada
        if (editorialSeleccionadaId > 0)
        {
            var ed = editoriales.FirstOrDefault(e => e.Id == editorialSeleccionadaId);
            if (ed != null)
            {
                libroActual.Editorial = ed;
            }
        }

        // Asignar categoría seleccionada
        if (!string.IsNullOrWhiteSpace(categoriaSeleccionada))
        {
            if (libroActual.Categoriases.Count > 0)
            {
                libroActual.Categoriases[0].NombreCategoria = categoriaSeleccionada;
            }
            else
            {
                libroActual.Categoriases.Add(new CategoriaDto { NombreCategoria = categoriaSeleccionada, Activo = true });
            }
        }

        bool ok;
        if (libroActual.Id == null || libroActual.Id == 0)
        {
            ok = await LibroService.Create(libroActual);
            mensajeAlerta = ok ? "Título registrado con éxito en el catálogo editorial." : "Error al registrar el título.";
        }
        else
        {
            ok = await LibroService.Update(libroActual);
            mensajeAlerta = ok ? "Título actualizado con éxito." : "Error al actualizar el título.";
        }
        guardando = false;

        if (ok)
        {
            CerrarModal();
            await CargarLibros();
        }
    }

    private void SolicitarEliminarLibro(LibroDto libro)
    {
        libroAEliminar = libro;
        mostrarModalEliminar = true;
    }

    private void CerrarModalEliminar()
    {
        libroAEliminar = null;
        mostrarModalEliminar = false;
    }

    private async Task ConfirmarEliminarLibro()
    {
        if (libroAEliminar?.Id != null)
        {
            var ok = await LibroService.Delete(libroAEliminar.Id.Value);
            mensajeAlerta = ok ? $"El título '{libroAEliminar.Titulo}' fue retirado del catálogo." : "Error al retirar el título.";
            CerrarModalEliminar();
            await CargarLibros();
        }
    }
}

    private LicenciaDigitalDto? licenciaAEliminar = null;
    private long libroSeleccionadoId = 0;
    private bool cargando = true;
    private bool mostrarModal = false;
    private bool mostrarModalEliminar = false;
    private string busquedaTexto = "";
    private string filtroEstado = "TODOS";
    private string? mensajeAlerta = null;

    private List<LicenciaDigitalDto> licenciasFiltradas => licencias
        .Where(l => filtroEstado == "TODOS" || l.Estado == filtroEstado)
        .Where(l => string.IsNullOrWhiteSpace(busquedaTexto) ||
                    l.CodigoLicencia.Contains(busquedaTexto, StringComparison.OrdinalIgnoreCase) ||
                    (l.Libro?.Titulo?.Contains(busquedaTexto, StringComparison.OrdinalIgnoreCase) ?? false) ||
                    (l.UsuarioCreacion?.Contains(busquedaTexto, StringComparison.OrdinalIgnoreCase) ?? false))
        .ToList();

    protected override async Task OnInitializedAsync()
    {
        await CargarDatos();
    }

    private async Task CargarDatos()
    {
        cargando = true;
        try
        {
            licencias = await LicenciaService.GetAll();
            catalogoLibros = await LibroService.GetAll();
        }
        catch (Exception ex)
        {
            mensajeAlerta = $"Error al cargar licencias: {ex.Message}";
        }
        finally
        {
            cargando = false;
        }
    }

    private void AbrirModalNuevaLicencia()
    {
        nuevaLicencia = new LicenciaDigitalDto
        {
            Estado = "DISPONIBLE",
            FechaCreacion = DateTime.UtcNow
        };
        GenerarCodigoAleatorio();
        if (catalogoLibros.Count > 0)
        {
            libroSeleccionadoId = catalogoLibros[0].Id ?? 0;
            nuevaLicencia.EnlaceDescarga = $"https://visor.bookstore.com/read/{libroSeleccionadoId}";
        }
        mostrarModal = true;
    }

    private void GenerarCodigoAleatorio()
    {
        nuevaLicencia.CodigoLicencia = $"LIC-{DateTime.UtcNow.Year}-{Guid.NewGuid().ToString("N")[..8].ToUpper()}";
    }

    private void CerrarModal()
    {
        mostrarModal = false;
    }

    private async Task GuardarLicencia()
    {
        if (string.IsNullOrWhiteSpace(nuevaLicencia.CodigoLicencia))
        {
            mensajeAlerta = "El código de licencia es obligatorio.";
            return;
        }

        var libro = catalogoLibros.FirstOrDefault(l => l.Id == libroSeleccionadoId);
        nuevaLicencia.Libro = libro;

        if (string.IsNullOrWhiteSpace(nuevaLicencia.EnlaceDescarga))
        {
            nuevaLicencia.EnlaceDescarga = $"https://visor.bookstore.com/read/{libroSeleccionadoId}";
        }

        var exito = await LicenciaService.Crear(nuevaLicencia);
        if (exito)
        {
            mensajeAlerta = $"Licencia '{nuevaLicencia.CodigoLicencia}' creada con éxito.";
            mostrarModal = false;
            await CargarDatos();
        }
        else
        {
            mensajeAlerta = "No se pudo crear la licencia. Revisa la conexión.";
        }
    }

    private void SolicitarEliminarLicencia(LicenciaDigitalDto lic)
    {
        licenciaAEliminar = lic;
        mostrarModalEliminar = true;
    }

    private void CerrarModalEliminar()
    {
        licenciaAEliminar = null;
        mostrarModalEliminar = false;
    }

    private async Task ConfirmarEliminarLicencia()
    {
        if (licenciaAEliminar?.Id.HasValue == true)
        {
            var exito = await LicenciaService.Eliminar(licenciaAEliminar.Id.Value);
            if (exito)
            {
                mensajeAlerta = $"Licencia '{licenciaAEliminar.CodigoLicencia}' eliminada correctamente.";
                CerrarModalEliminar();
                await CargarDatos();
            }
            else
            {
                mensajeAlerta = "Error al intentar eliminar la licencia.";
            }
        }
    }

namespace BlazorApp1.Components.Services;

public class BackendStatusService
{
    private readonly HttpClient _http;

    public bool ModoMemoriaRAM { get; private set; } = true;
    public bool BackendEnLinea { get; private set; } = false;
    public string UltimoMensaje { get; private set; } = "Modo Local Activo";

    public event Action? OnChange;

    public BackendStatusService(IHttpClientFactory httpClientFactory)
    {
        _http = httpClientFactory.CreateClient("SpringBoot");
    }

    public void ActivarModoRAM()
    {
        ModoMemoriaRAM = true;
        BackendEnLinea = false;
        UltimoMensaje = "Modo Local Activo";
        NotifyStateChanged();
    }

    public async Task<bool> ProbarConexionBackendAsync()
    {
        try
        {
            using var cts = new CancellationTokenSource(TimeSpan.FromMilliseconds(1000));
            var res = await _http.GetAsync("management/health", cts.Token);
            if (res.IsSuccessStatusCode)
            {
                BackendEnLinea = true;
                ModoMemoriaRAM = false;
                UltimoMensaje = "Conectado al Servidor API";
                NotifyStateChanged();
                return true;
            }
        }
        catch
        {
            BackendEnLinea = false;
            ModoMemoriaRAM = true;
            UltimoMensaje = "Servidor no disponible. Continuando en modo local.";
            NotifyStateChanged();
        }

        return false;
    }

    private void NotifyStateChanged() => OnChange?.Invoke();
}

namespace BlazorApp1.Components.Services;

public class CarritoItem
{
    public LibroDto Libro { get; set; } = new();
    public int Cantidad { get; set; } = 1;
    public decimal Subtotal => Libro.Precio * Cantidad;
}

public class CarritoService
{
    public List<CarritoItem> Items { get; private set; } = new();
    public PromocionDto? PromocionAplicada { get; private set; }

    public event Action? OnChange;

    public int TotalItems => Items.Sum(i => i.Cantidad);
    public decimal SubtotalBruto => Items.Sum(i => i.Subtotal);

    public decimal MontoDescuento => PromocionAplicada != null
        ? Math.Round(SubtotalBruto * (decimal)(PromocionAplicada.PorcentajeDescuento / 100.0f), 2)
        : 0m;

    public decimal TotalPagar => Math.Max(0m, SubtotalBruto - MontoDescuento);

    public void AgregarLibro(LibroDto libro)
    {
        var existente = Items.FirstOrDefault(i => i.Libro.Id == libro.Id);
        if (existente != null)
        {
            existente.Cantidad++;
        }
        else
        {
            Items.Add(new CarritoItem { Libro = libro, Cantidad = 1 });
        }
        NotifyStateChanged();
    }

    public void ModificarCantidad(long libroId, int delta)
    {
        var item = Items.FirstOrDefault(i => i.Libro.Id == libroId);
        if (item != null)
        {
            item.Cantidad += delta;
            if (item.Cantidad <= 0)
            {
                Items.Remove(item);
            }
            NotifyStateChanged();
        }
    }

    public void EliminarItem(long libroId)
    {
        var item = Items.FirstOrDefault(i => i.Libro.Id == libroId);
        if (item != null)
        {
            Items.Remove(item);
            NotifyStateChanged();
        }
    }

    public void AplicarPromocion(PromocionDto? promo)
    {
        PromocionAplicada = promo;
        NotifyStateChanged();
    }

    public void RemoverPromocion()
    {
        PromocionAplicada = null;
        NotifyStateChanged();
    }

    public void Limpiar()
    {
        Items.Clear();
        PromocionAplicada = null;
        NotifyStateChanged();
    }

    private void NotifyStateChanged() => OnChange?.Invoke();
}

    public async Task<List<CalificacionUsuarioDto>> GetAll()
    {
        if (_backendStatus.ModoMemoriaRAM)
        {
            return _calificacionesLocales.OrderByDescending(c => c.Id).ToList();
        }

        SetAuthorizationHeader();
        try
        {
            var result = await _http.GetFromJsonAsync<List<CalificacionUsuarioDto>>("api/calificacion-usuarios");
            if (result != null && result.Count > 0)
            {
                foreach (var item in result)

                    if (!_calificacionesLocales.Any(c => c.Id == item.Id))
                    {
                        _calificacionesLocales.Add(item);

        calificacion.FechaCalificacion = DateTime.UtcNow;
        calificacion.FechaCreacion ??= DateTime.UtcNow;
        calificacion.Activo = true;
        calificacion.UsuarioCreacion ??= _tokenService.CurrentUser?.Login ?? "cliente";

        try
        {
            var response = await _http.PostAsJsonAsync("api/calificacion-usuarios", calificacion);
            if (response.IsSuccessStatusCode)
            {
                var creada = await response.Content.ReadFromJsonAsync<CalificacionUsuarioDto>();
                if (creada != null)

        }

        calificacion.Id = (_calificacionesLocales.Count > 0 ? _calificacionesLocales.Max(c => c.Id ?? 0) : 0) + 1;
        _calificacionesLocales.Add(calificacion);
        return true;
    }

    }

}

public class LibroDto
{
    [JsonPropertyName("id")]
    public long? Id { get; set; }

    [JsonPropertyName("isbn")]
    public string Isbn { get; set; } = string.Empty;

    [JsonPropertyName("titulo")]
    public string Titulo { get; set; } = string.Empty;

    [JsonPropertyName("sinopsis")]
    public string Sinopsis { get; set; } = string.Empty;

    [JsonPropertyName("precio")]
    public decimal Precio { get; set; }

    [JsonPropertyName("idioma")]
    public string Idioma { get; set; } = "Español";

    [JsonPropertyName("activo")]
    public bool Activo { get; set; } = true;

    [JsonPropertyName("portadaUrl")]
    public string? PortadaUrl { get; set; }

    [JsonPropertyName("categoriases")]
    public List<CategoriaDto> Categoriases { get; set; } = new();

    [JsonPropertyName("autoreses")]
    public List<AutorDto> Autoreses { get; set; } = new();

    [JsonPropertyName("editorial")]
    public EditorialDto? Editorial { get; set; }

    [JsonPropertyName("promocioneses")]
    public List<PromocionDto> Promociones { get; set; } = new();

    // Propiedades de conveniencia y compatibilidad con vistas
    public string Autor
    {
        get => AutorPrincipal;
        set
        {
            var parts = (value ?? string.Empty).Trim().Split(' ', 2, StringSplitOptions.RemoveEmptyEntries);
            var nombres = parts.Length > 0 ? parts[0] : string.Empty;
            var apellidos = parts.Length > 1 ? parts[1] : string.Empty;

            if (Autoreses.Count > 0)
            {
                Autoreses[0].Nombres = nombres;
                Autoreses[0].Apellidos = apellidos;
            }
            else
            {
                Autoreses.Add(new AutorDto { Nombres = nombres, Apellidos = apellidos });
            }
        }
    }

    public int Stock { get; set; } = 999;

    public long? CategoriaId
    {
        get => Categoriases.FirstOrDefault()?.Id;
        set
        {
            if (value.HasValue)
            {
                if (Categoriases.Count > 0) Categoriases[0].Id = value;
                else Categoriases.Add(new CategoriaDto { Id = value });
            }
        }
    }

    [JsonIgnore]
    public string AutorPrincipal => Autoreses.Count > 0 ? Autoreses[0].NombreCompleto : "Autor de Catálogo";

    [JsonIgnore]
    public string CategoriaPrincipal => Categoriases.Count > 0 ? Categoriases[0].NombreCategoria : "Digital";

    [JsonIgnore]
    public string NombreEditorial => Editorial?.NombreEditorial ?? "Editorial Digital";

    [JsonIgnore]
    public PromocionDto? PromocionActiva => Promociones.FirstOrDefault(p => p.EsVigente);

    [JsonIgnore]
    public decimal PrecioFinal => PromocionActiva != null
        ? Math.Round(Precio * (decimal)(1 - (PromocionActiva.PorcentajeDescuento / 100.0f)), 2)
        : Precio;
}
